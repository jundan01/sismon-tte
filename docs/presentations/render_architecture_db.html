        bigint id PK "GENERATED ALWAYS AS IDENTITY"

        integer total_asn "Populasi ASN (Metrik Adopsi)"
        boolean is_active "Status Aktif"
        timestamptz created_at
        timestamptz updated_at
    }

    MASTER_JABATAN_GOLONGAN {
        bigint id PK "GENERATED ALWAYS AS IDENTITY"
        varchar kategori "CHECK IN ('JABATAN', 'GOLONGAN')"
        varchar kode UK "Kode Referensi Unik"
        varchar nama_label "Label Jabatan / Golongan"
        timestamptz created_at
    }

    USERS {
        bigint id PK "GENERATED ALWAYS AS IDENTITY"
        varchar name "Nama Lengkap Superadmin"
        varchar email UK "Email Dinas Kredensial"
        varchar password "Bcrypt Hash"
        text mfa_secret "AES-256-GCM Encrypted TOTP Secret"
        text mfa_recovery_codes "Encrypted Backup Recovery Codes"
        boolean is_mfa_enabled "Flag MFA Aktif"
        varchar role "CHECK = 'superadmin'"
        varchar last_login_ip "IP VPN Client"
        timestamptz last_login_at "Waktu Login Terakhir"
        timestamptz created_at
        timestamptz updated_at
    }

    PENDAFTARAN_TTE {
        bigint id PK "GENERATED ALWAYS AS IDENTITY"
        varchar nomor_referensi UK "Ref Registrasi: TTE-YYYYMMDD-XXXXXX"
        varchar nama_lengkap "Nama ASN Murni Tanpa Gelar"
        varchar nik "AES-256-CBC Deterministik (Reversible)"
        varchar nip "AES-256-CBC Deterministik (Reversible)"
        varchar no_hp "Nomor WhatsApp Kontak"
        bigint skpd_id FK "Relasi ke skpd.id"
        bigint jabatan_id FK "Relasi ke master_jabatan_golongan.id"
        bigint golongan_id FK "Relasi ke master_jabatan_golongan.id"
        boolean status_memiliki_email "Sudah Ada Email / Belum"
        varchar email_existing "Email Existing Jika Ada"
        varchar email_usulan "Email Usulan Auto-Generated"
        varchar email_final_terverifikasi "Email Definitif Terkunci"
        boolean is_email_recap_confirmed "Flag Rekap Email Terkunci"
        varchar status "CHECK: MENUNGGU_VERIFIKASI, DISETUJUI, DITOLAK"
        text catatan_penolakan "Catatan Jika Ditolak"

        timestamptz verified_at "Waktu Persetujuan Tahap 1"
        varchar verified_ip "IP Verifikator Tahap 1"
        timestamptz created_at
        timestamptz updated_at
    }

        bigint id PK "GENERATED ALWAYS AS IDENTITY"
